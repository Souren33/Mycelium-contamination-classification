#Normalizing each color bin separately. Ie if red total count is 130, and one bin has 20 then we get a value of 20/130 for that bin.
normalise_hist <- function(df, feature_cols) {
  df_norm <- df

  r_cols <- grep("^r_bins", feature_cols, value = TRUE)
  g_cols <- grep("^g_bins", feature_cols, value = TRUE)
  b_cols <- grep("^b_bins", feature_cols, value = TRUE)

  for (cols in list(r_cols, g_cols, b_cols)) {
    chunk    <- as.matrix(df[, cols])
    row_sums <- rowSums(chunk)
    row_sums[row_sums == 0] <- 1 #No div by 0
    df_norm[, cols] <- chunk / row_sums
  }
  df_norm
}

#Expect mean roughly centered at .5 via beta, gamma for tau demonstarates we are uncertain
a_mu  <- 2;  b_mu  <- 2
a_tau <- 2;  b_tau <- 1
# Using log to avoid multiplication
log_prior <- function(mu, tau) {
  if (mu <= 0 || mu >= 1 || tau <= 0) return(-Inf)
  dbeta(mu,   a_mu,  b_mu,  log = TRUE) +
  dgamma(tau, a_tau, b_tau, log = TRUE)
}
#Gaussian stdev 1/sqrt(tau)
#Mult with log becomes addition
log_likelihood_feat <- function(x, mu, tau) {
  sum(dnorm(x, mean = mu, sd = 1 / sqrt(tau), log = TRUE))
}

log_posterior_feat <- function(x, mu, tau) {
  log_likelihood_feat(x, mu, tau) + log_prior(mu, tau)
}
mh_sampler <- function(x, n_iter = 2000, burnin = 500, thin = 5,
                       prop_sd_mu = 0.01, prop_sd_tau = 0.1) {
  mu_cur  <- mean(x)
  tau_cur <- 1 / var(x)
  if (!is.finite(tau_cur)) tau_cur <- 1 #Set to one if var(x) = 0(all vals same)
  mu_cur <- min(max(mu_cur, 1e-6), 1 - 1e-6) #prevent mu from hitting 1 and 0 for beta issues

  samples <- matrix(NA, nrow = n_iter, ncol = 2,
                    dimnames = list(NULL, c("mu", "tau")))
  lp_cur <- log_posterior_feat(x, mu_cur, tau_cur)

  for (i in seq_len(n_iter)) {
    mu_prop  <- mu_cur  + rnorm(1, 0, prop_sd_mu)
    tau_prop <- tau_cur * exp(rnorm(1, 0, prop_sd_tau))
    lp_prop  <- log_posterior_feat(x, mu_prop, tau_prop)

    if (is.finite(lp_prop) && is.finite(lp_cur)) {
      log_accept <- lp_prop - lp_cur + log(tau_prop) - log(tau_cur) #Check if value shows improvement ie positive result(won't be finite if nt)
      if (is.finite(log_accept) && log(runif(1)) < log_accept) {
        mu_cur  <- mu_prop
        tau_cur <- tau_prop
        lp_cur  <- lp_prop
      }
    }#Record regardless accept/reject proposed
    samples[i, ] <- c(mu_cur, tau_cur)
  }
  keep_idx <- seq(burnin + 1, n_iter, by = thin)
  samples[keep_idx, ]
}

compute_posterior_samples <- function(df_train, feature_cols, class_priors) {
  classes   <- names(class_priors)
  all_samps <- list()
  total     <- length(feature_cols) * length(classes)
  done      <- 0

  for (cl in classes) {
    subset <- df_train |>
      filter(as.character(label) == as.character(cl)) |>  # safe string match
      select(all_of(feature_cols)) |>
      as.matrix()

    all_samps[[cl]] <- lapply(seq_len(ncol(subset)), function(j) {
      done <<- done + 1
      cat(sprintf("\r  Chain %d / %d", done, total))
      flush.console()
      mh_sampler(subset[, j])
    })
  }
  cat("\n")
  all_samps
}

#Each loop s becomes new row of values
log_predictive <- function(x_new, samples) {
  log_liks <- apply(samples, 1, function(s) {
    dnorm(x_new, mean = s["mu"], sd = 1 / sqrt(s["tau"]), log = TRUE)
  })
  max_ll <- max(log_liks)
  max_ll + log(mean(exp(log_liks - max_ll)))
}

predict_one <- function(obs, posterior_samples, class_priors) {
  log_posteriors <- sapply(names(class_priors), function(cl) {
    samps   <- posterior_samples[[cl]]
    log_lik <- sum(mapply(log_predictive, obs, samps))
    log(class_priors[cl]) + log_lik
  })
  as.integer(names(which.max(log_posteriors)))
}
#predicts for each row in the DF
predict_bayes <- function(df_new, feature_cols, posterior_samples, class_priors) {
  X <- as.matrix(df_new[, feature_cols])
  apply(X, 1, predict_one,
        posterior_samples = posterior_samples,
        class_priors      = class_priors)
}

compute_metrics <- function(conf_mat, classes) {
  metrics <- lapply(as.character(classes), function(cl) {
    TP <- conf_mat[cl, cl]
    FP <- sum(conf_mat[cl, ]) - TP
    FN <- sum(conf_mat[, cl]) - TP
    TN <- sum(conf_mat) - TP - FP - FN

    precision <- if ((TP + FP) == 0) 0 else TP / (TP + FP)
    recall    <- if ((TP + FN) == 0) 0 else TP / (TP + FN)
    f1        <- if ((precision + recall) == 0) 0 else
                   2 * precision * recall / (precision + recall)
    list(class = cl, TP = TP, FP = FP, FN = FN, TN = TN,
         precision = precision, recall = recall, f1 = f1)
  })
  bind_rows(lapply(metrics, as.data.frame))
}
